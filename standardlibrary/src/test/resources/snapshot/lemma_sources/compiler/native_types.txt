zero_scheme: zero = zero ∥ nat_type + zero
successor_scheme: successor = successor ∥ arrow nat_type nat_type + successor
true_scheme: true = true ∥ boolean_type + true
false_scheme: false = false ∥ boolean_type + false
boolean_not_scheme: boolean_not = boolean_not ∥ arrow boolean_type boolean_type + boolean_not
nil_scheme: nil = nil ∥ forall λx_0. list_type x_0 + nil
cons_scheme: cons = cons ∥ forall λx_0. arrow x_0 λ. arrow (list_type x_0) λ. list_type x_0 + cons
identity_scheme: identity = identity ∥ forall λx_0. arrow x_0 x_0 + identity
konstant_scheme: konstant = konstant ∥ forall λx_0. forall λx_1. arrow x_0 λ. arrow x_1 x_0 + konstant
compose_scheme: compose = compose
          ∥ forall λx_0. forall λx_1. forall λx_2. arrow (arrow x_1 x_2) λ. arrow (arrow x_0 x_1) λ. arrow x_0 x_2
          + compose
fixpoint_scheme: fixpoint = fixpoint ∥ forall λx_0. arrow (arrow x_0 x_0) x_0 + fixpoint
nat_fold_scheme: nat_fold = nat_fold ∥ forall λx_0. arrow x_0 λ. arrow (arrow x_0 x_0) λ. arrow nat_type x_0 + nat_fold
list_fold_scheme: list_fold = list_fold
          ∥ forall λx_0. forall λx_1. arrow x_1 λ. arrow (arrow x_0 λ. arrow x_1 x_1) λ. arrow (list_type x_0) x_1
          + list_fold
